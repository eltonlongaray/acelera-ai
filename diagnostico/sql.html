SQL do Diagnóstico (Acelera Aí)

  1. Abre o Supabase → projeto realeza-holistica (o mesmo do castelo).
  2. SQL Editor → New query.
  3. Toca em Copiar SQL, cola lá e clica em Run.
  4. Tem que aparecer "Success. No rows returned". Pode rodar de novo sem problema.

create table if not exists public.acelera_diagnostico (
  id bigint generated always as identity primary key,
  criado_em timestamptz not null default now(),
  token text unique,
  cliente text,
  previas jsonb not null default '{}'::jsonb,
  respostas jsonb,
  enviado_em timestamptz,
  proposta jsonb
);
alter table public.acelera_diagnostico enable row level security;

drop policy if exists "formulario padrao envia" on public.acelera_diagnostico;
create policy "formulario padrao envia" on public.acelera_diagnostico
  for insert to anon, authenticated
  with check (token is null and previas = '{}'::jsonb and proposta is null
              and length(coalesce(respostas::text,'')) < 20000);

drop policy if exists "admin tudo" on public.acelera_diagnostico;

-- Link de cliente: devolve só o nome e QUAIS perguntas pular (nunca as respostas)
create or replace function public.acelera_diag_abrir(p_token text)
returns jsonb language sql security definer set search_path = public as $$
  select jsonb_build_object(
    'nome', previas->>'nome',
    'pular', (select coalesce(jsonb_agg(k), '[]'::jsonb) from jsonb_object_keys(previas) k),
    'enviado', enviado_em is not null)
  from public.acelera_diagnostico where token = p_token;
$$;

-- Link de cliente: grava as respostas uma vez só
create or replace function public.acelera_diag_enviar(p_token text, p_respostas jsonb)
returns boolean language plpgsql security definer set search_path = public as $$
begin
  if length(p_respostas::text) > 20000 then return false; end if;
  update public.acelera_diagnostico set respostas = p_respostas, enviado_em = now()
   where token = p_token and enviado_em is null;
  return found;
end $$;

grant execute on function public.acelera_diag_abrir(text) to anon, authenticated;
grant execute on function public.acelera_diag_enviar(text, jsonb) to anon, authenticated;
